function renderIntegrations(){document.getElementById('intGrid').innerHTML=integrations.map((x,i)=>{const on=localStorage.getItem('int-'+i)==='1';return `<div class="card integration"><div class="inttop"><div class="intlogo">${x[1]}</div><div><div class="intname">${x[0]}</div><div class="intdesc">${x[3]}</div></div></div><div class="intstatus ${on?'on':''}">${on?'● Подключено · синхронизация активна':'○ Не подключено'}</div><div class="intactions"><span class="mini">${x[2]}</span><button class="btn ${on?'':'primary'}" onclick="toggleInt(${i})">${on?'Настроить':'Подключить'}</button></div></div>`}).join('')}

function openModal(){document.getElementById('modal').classList.add('open')}function closeModal(){document.getElementById('modal').classList.remove('open')}document.getElementById('modal').onclick=e=>{if(e.target.id==='modal')closeModal()}

function createCandidate(){const name=document.getElementById('ncname').value.trim();if(!name){toast('Укажите имя кандидата');return}data.candidates.unshift({id:Date.now(),name,role:document.getElementById('ncrole').value||'Специалист',vac:document.getElementById('ncvac').value,stage:'Отклики',rating:0,source:document.getElementById('ncsource').value,tags:[],email:document.getElementById('ncemail').value,phone:document.getElementById('ncphone').value});save();closeModal();renderCandidates();renderPipeline();toast('Кандидат добавлен')}

function changeStage(id,s){const c=data.candidates.find(x=>x.id===id);c.stage=s;save();renderCandidates();renderPipeline();toast(`Этап изменён: ${s}`)}
